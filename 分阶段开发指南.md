        assert!(table.remove("/test"));
        assert!(!table.remove("/nonexistent"));

    #[test]
    fn test_concurrent_access() {
        use std::sync::Arc;
        let table = Arc::new(RouteTable::new());
        let mut handles = vec![];

        for i in 0..10 {
            let table_clone = Arc::clone(&table);
            let handle = std::thread::spawn(move || {
                let route = SimpleRoute::new(&format!("body-{}", i), "text/plain");
                table_clone.insert(&format!("/path-{}", i), Box::new(route));
            });
            handles.push(handle);
        }

        for handle in handles {
            handle.join().unwrap();
        }

        assert_eq!(table.get("/path-0").is_some(), true);
        assert_eq!(table.get("/path-9").is_some(), true);
    }

## 存储接口设计说明

### 为什么需要简单的 CRUD 接口？

1. **开箱即用**：用户可以直接使用文件存储或内存存储，无需自己实现读写逻辑
2. **统一抽象**：`KeyValueStorage` trait 提供统一的接口，便于切换存储后端
3. **减少重复代码**：常见操作（读取配置、保存状态、缓存数据）都有现成的方法
4. **测试友好**：`MemoryStorage` 可以轻松替换文件存储用于单元测试

### 接口对比

| 存储类型 | 读取 | 写入 | 删除 | 存在检查 | 适用场景 |
|---------|------|------|------|---------|---------|
| FileStorage | ✅ | ✅ | ✅ | ✅ | 配置文件、日志、持久化数据 |
| MemoryStorage | ✅ | ✅ | ✅ | ✅ | 缓存、临时数据、测试环境 |
| DatabaseStorage | ✅ | ✅ | ✅ | ✅ | 生产环境、大规模数据 |
