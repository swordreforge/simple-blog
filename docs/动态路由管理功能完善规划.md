```rust
// 路由存储抽象层
pub trait RouteStorage {
    async fn save_route(&self, route: &DynamicRoute) -> Result<(), StorageError>;
    async fn load_route(&self, id: i64) -> Result<Option<DynamicRoute>, StorageError>;
    async fn delete_route(&self, id: i64) -> Result<(), StorageError>;
    async fn list_routes(&self) -> Result<Vec<DynamicRoute>, StorageError>;
}

// Memory存储实现
pub struct MemoryRouteStorage {
    routes: Arc<RwLock<HashMap<i64, DynamicRoute>>>,
}

// File存储实现
pub struct FileRouteStorage {
    base_dir: PathBuf,
}

impl FileRouteStorage {
    pub fn new(base_dir: &str) -> Result<Self, StorageError> {
        let base_path = PathBuf::from(base_dir);
        // 创建基础目录
        fs::create_dir_all(&base_path)?;
        // 创建routes子目录
        let routes_dir = base_path.join("routes");
        fs::create_dir_all(&routes_dir)?;
        Ok(Self { base_dir: base_path })
    }

    fn get_route_file_path(&self, id: i64) -> PathBuf {
        self.base_dir.join("routes").join(format!("route_{}.json", id))
    }

    async fn save_route(&self, route: &DynamicRoute) -> Result<(), StorageError> {
        let file_path = self.get_route_file_path(route.id.unwrap_or(0));
        let content = serde_json::to_string_pretty(route)?;
        fs::write(&file_path, content)?;
        Ok(())
    }

    async fn load_route(&self, id: i64) -> Result<Option<DynamicRoute>, StorageError> {
        let file_path = self.get_route_file_path(id);
        if !file_path.exists() {
            return Ok(None);
        }
        let content = fs::read_to_string(&file_path)?;
        let route: DynamicRoute = serde_json::from_str(&content)?;
        Ok(Some(route))
    }

    async fn delete_route(&self, id: i64) -> Result<(), StorageError> {
        let file_path = self.get_route_file_path(id);
        if file_path.exists() {
            fs::remove_file(&file_path)?;
        }
        Ok(())
    }

    async fn list_routes(&self) -> Result<Vec<DynamicRoute>, StorageError> {
        let routes_dir = self.base_dir.join("routes");
        let mut routes = Vec::new();

        for entry in fs::read_dir(&routes_dir)? {
            let entry = entry?;
            let path = entry.path();
            if path.extension().and_then(|s| s.to_str()) == Some("json") {
                let content = fs::read_to_string(&path)?;
                let route: DynamicRoute = serde_json::from_str(&content)?;
                routes.push(route);
            }
        }

        Ok(routes)
    }
}
```

**目录操作函数**：
```rust
use std::fs;
use std::path::PathBuf;

pub struct RouteFileManager {
    base_dir: PathBuf,
    routes_dir: PathBuf,
    backups_dir: PathBuf,
    templates_dir: PathBuf,
}

impl RouteFileManager {
    pub fn new(base_dir: &str) -> Result<Self, std::io::Error> {
        let base_path = PathBuf::from(base_dir);
        
        // 创建所有必要的目录
        let routes_dir = base_path.join("routes");
        let backups_dir = base_path.join("backups");
        let templates_dir = base_path.join("templates");
        
        fs::create_dir_all(&routes_dir)?;
        fs::create_dir_all(&backups_dir)?;
        fs::create_dir_all(&templates_dir)?;
        
        // 创建.gitignore文件，避免提交路由配置到版本控制
        let gitignore_path = base_path.join(".gitignore");
        if !gitignore_path.exists() {
            fs::write(&gitignore_path, "# 路由配置文件\n*.json\n!templates/\n")?;
        }
        
        Ok(Self {
            base_dir: base_path,
            routes_dir,
            backups_dir,
            templates_dir,
        })
    }
    
    // 验证文件路径安全性
    pub fn validate_path(&self, path: &str) -> Result<(), StorageError> {
        let path = PathBuf::from(path);
        
        // 防止路径遍历攻击
        if path.has_root() || path.starts_with("..") {
            return Err(StorageError::InvalidPath("Invalid path".to_string()));
        }
        
        // 确保路径在base_dir内
        let full_path = self.base_dir.join(&path);
        if !full_path.starts_with(&self.base_dir) {
            return Err(StorageError::InvalidPath("Path outside base directory".to_string()));
        }
        
        Ok(())
    }
    
    // 创建路由文件
    pub fn create_route_file(&self, route: &DynamicRoute) -> Result<PathBuf, StorageError> {
        let id = route.id.ok_or(StorageError::InvalidData("Route ID required".to_string()))?;
        let file_path = self.routes_dir.join(format!("route_{}.json", id));
        
        // 验证文件路径
        self.validate_path(file_path.to_str().unwrap())?;
        
        // 写入文件
        let content = serde_json::to_string_pretty(route)
            .map_err(|e| StorageError::SerializationError(e.to_string()))?;
        
        fs::write(&file_path, content)
            .map_err(|e| StorageError::FileError(e.to_string()))?;
        
        Ok(file_path)
    }
    
    // 读取路由文件
    pub fn read_route_file(&self, id: i64) -> Result<Option<DynamicRoute>, StorageError> {
        let file_path = self.routes_dir.join(format!("route_{}.json", id));
        
        if !file_path.exists() {
            return Ok(None);
        }
        
        let content = fs::read_to_string(&file_path)
            .map_err(|e| StorageError::FileError(e.to_string()))?;
        
        let route: DynamicRoute = serde_json::from_str(&content)
            .map_err(|e| StorageError::SerializationError(e.to_string()))?;
        
        Ok(Some(route))
    }
    
    // 删除路由文件
    pub fn delete_route_file(&self, id: i64) -> Result<(), StorageError> {
        let file_path = self.routes_dir.join(format!("route_{}.json", id));
        
        if file_path.exists() {
            fs::remove_file(&file_path)
                .map_err(|e| StorageError::FileError(e.to_string()))?;
        }
        
        Ok(())
    }
    
    // 备份路由文件
    pub fn backup_route_file(&self, id: i64) -> Result<PathBuf, StorageError> {
        let source_path = self.routes_dir.join(format!("route_{}.json", id));
        let timestamp = chrono::Utc::now().format("%Y%m%d_%H%M%S");
        let backup_path = self.backups_dir.join(format!("route_{}_backup_{}.json", id, timestamp));
        
        if source_path.exists() {
            fs::copy(&source_path, &backup_path)
                .map_err(|e| StorageError::FileError(e.to_string()))?;
        }
        
        Ok(backup_path)
    }
    
    // 列出所有路由文件
    pub fn list_route_files(&self) -> Result<Vec<i64>, StorageError> {
        let mut route_ids = Vec::new();
        
        for entry in fs::read_dir(&self.routes_dir)
            .map_err(|e| StorageError::FileError(e.to_string()))?
        {
            let entry = entry.map_err(|e| StorageError::FileError(e.to_string()))?;
            let path = entry.path();
            
            if path.extension().and_then(|s| s.to_str()) == Some("json") {
                // 从文件名提取ID：route_123.json -> 123
                let file_name = path.file_stem()
                    .and_then(|s| s.to_str())
                    .ok_or_else(|| StorageError::InvalidData("Invalid file name".to_string()))?;
                
                if let Some(id_str) = file_name.strip_prefix("route_") {
                    if let Ok(id) = id_str.parse::<i64>() {
                        route_ids.push(id);
                    }
                }
            }
        }
        
        route_ids.sort();
        Ok(route_ids)
    }
    
    // 清理过期备份文件
    pub fn cleanup_old_backups(&self, keep_count: usize) -> Result<Vec<PathBuf>, StorageError> {
        let mut removed_files = Vec::new();
        
        // 按路由ID分组备份文件
        let mut backups_by_route: std::collections::HashMap<i64, Vec<std::fs::DirEntry>> = 
            std::collections::HashMap::new();
        
        for entry in fs::read_dir(&self.backups_dir)
            .map_err(|e| StorageError::FileError(e.to_string()))?
        {
            let entry = entry.map_err(|e| StorageError::FileError(e.to_string()))?;
            let path = entry.path();
            
            if let Some(file_name) = path.file_name().and_then(|s| s.to_str()) {
                // 解析文件名：route_123_backup_20260317_100000.json
                if let Some(captures) = regex::Regex::new(r"route_(\d+)_backup_\d+_\d+\.json")
                    .unwrap()
                    .captures(file_name)
                {
                    if let Ok(id) = captures[1].parse::<i64>() {
                        backups_by_route.entry(id).or_insert_with(Vec::new).push(entry);
                    }
                }
            }
        }
        
        // 对每个路由的备份按时间排序，保留最新的N个
        for (_, mut backups) in backups_by_route {
            if backups.len() > keep_count {
                // 按修改时间排序
                backups.sort_by(|a, b| {
                    let a_time = a.metadata().and_then(|m| m.modified()).unwrap_or(std::time::SystemTime::UNIX_EPOCH);
                    let b_time = b.metadata().and_then(|m| m.modified()).unwrap_or(std::time::SystemTime::UNIX_EPOCH);
                    b_time.cmp(&a_time)
                });
                
                // 删除多余的备份
                for backup in backups.iter().skip(keep_count) {
                    let path = backup.path();
                    fs::remove_file(&path)
                        .map_err(|e| StorageError::FileError(e.to_string()))?;
                    removed_files.push(path);
                }
            }
        }
        
        Ok(removed_files)
    }
}
```

```rust
use std::collections::HashMap;
use std::sync::{Arc, RwLock};
use tokio::time::{interval, Duration};

pub struct MemoryRouteStorage {
    routes: Arc<RwLock<HashMap<i64, DynamicRoute>>>,
    max_routes: usize,
    cleanup_interval: Duration,
}

impl MemoryRouteStorage {
    pub fn new(max_routes: usize, cleanup_interval_secs: u64) -> Self {
        let storage = Self {
            routes: Arc::new(RwLock::new(HashMap::new())),
            max_routes,
            cleanup_interval: Duration::from_secs(cleanup_interval_secs),
        };
        
        // 启动后台清理任务
        storage.start_cleanup_task();
        
        storage
    }
    
    fn start_cleanup_task(&self) {
        let routes = Arc::clone(&self.routes);
        let interval = self.cleanup_interval;
        
        tokio::spawn(async move {
            let mut interval_timer = interval(interval);
            loop {
                interval_timer.tick().await;
                
                // 定期清理：可以添加自动清理过期路由的逻辑
                if let Ok(mut routes) = routes.write() {
                    // 示例：清理已禁用的路由（可选功能）
                    // routes.retain(|_, route| route.enabled);
                    
                    // 检查路由数量限制
                    if routes.len() > Self::MAX_ROUTES_DEFAULT {
                        // 按创建时间排序，删除最旧的
                        let mut route_list: Vec<_> = routes.iter().collect();
                        route_list.sort_by(|a, b| a.1.created_at.cmp(&b.1.created_at));
                        
                        let to_remove = route_list.len() - Self::MAX_ROUTES_DEFAULT;
                        for (id, _) in route_list.iter().take(to_remove) {
                            routes.remove(id);
                        }
                    }
                }
            }
        });
    }
    
    async fn save_route(&self, route: &DynamicRoute) -> Result<(), StorageError> {
        let mut routes = self.routes.write().await;
        
        // 检查路由数量限制
        if routes.len() >= self.max_routes && !routes.contains_key(&route.id.unwrap_or(0)) {
            return Err(StorageError::CapacityExceeded(
                format!("Maximum routes limit {} reached", self.max_routes)
            ));
        }
        
        let id = route.id.ok_or(StorageError::InvalidData("Route ID required".to_string()))?;
        routes.insert(id, route.clone());
        
        Ok(())
    }
    
    async fn load_route(&self, id: i64) -> Result<Option<DynamicRoute>, StorageError> {
        let routes = self.routes.read().await;
        Ok(routes.get(&id).cloned())
    }
    
    async fn delete_route(&self, id: i64) -> Result<(), StorageError> {
        let mut routes = self.routes.write().await;
        routes.remove(&id);
        Ok(())
    }
    
    async fn list_routes(&self) -> Result<Vec<DynamicRoute>, StorageError> {
        let routes = self.routes.read().await;
        Ok(routes.values().cloned().collect())
    }
    
    async fn clear_all(&self) -> Result<(), StorageError> {
        let mut routes = self.routes.write().await;
        routes.clear();
        Ok(())
    }
    
    async fn get_stats(&self) -> RouteStorageStats {
        let routes = self.routes.read().await;
        RouteStorageStats {
            total_routes: routes.len(),
            enabled_routes: routes.values().filter(|r| r.enabled).count(),
            disabled_routes: routes.values().filter(|r| !r.enabled).count(),
            memory_usage_bytes: std::mem::size_of_val(&*routes),
        }
    }
```

### 10.4 路由类型管理器

```rust
pub struct RouteTypeManager {
    database_storage: Arc<DatabaseRouteStorage>,
    file_storage: Arc<FileRouteStorage>,
    memory_storage: Arc<MemoryRouteStorage>,
    default_type: RouteType,
}

impl RouteTypeManager {
    pub fn new(
        db_storage: Arc<DatabaseRouteStorage>,
        file_storage: Arc<FileRouteStorage>,
        memory_storage: Arc<MemoryRouteStorage>,
        default_type: RouteType,
    ) -> Self {
        Self {
            database_storage: db_storage,
            file_storage: file_storage,
            memory_storage: memory_storage,
            default_type,
        }
    }
    
    pub fn get_storage(&self, route_type: &RouteType) -> Arc<dyn RouteStorage> {
        match route_type {
            RouteType::Database => Arc::clone(&self.database_storage) as Arc<dyn RouteStorage>,
            RouteType::File => Arc::clone(&self.file_storage) as Arc<dyn RouteStorage>,
            RouteType::Memory => Arc::clone(&self.memory_storage) as Arc<dyn RouteStorage>,
        }
    }
    
    pub async fn save_route(&self, mut route: DynamicRoute) -> Result<i64, StorageError> {
        // 如果没有指定路由类型，使用默认类型
        let route_type = route.route_type.clone().unwrap_or(self.default_type.clone());
        
        // 根据路由类型选择存储
        let storage = self.get_storage(&route_type);
        
        // 更新路由类型
        route.route_type = Some(route_type);
        
        // 保存路由
        storage.save_route(&route).await?;
        
        Ok(route.id.unwrap_or(0))
    }
    
    pub async fn load_route(&self, id: i64, route_type: Option<RouteType>) -> Result<Option<DynamicRoute>, StorageError> {
        // 如果指定了路由类型，从指定类型加载
        if let Some(route_type) = route_type {
            let storage = self.get_storage(&route_type);
            return storage.load_route(id).await;
        }
        
        // 否则依次从各存储类型查找
        for storage_type in [RouteType::Database, RouteType::File, RouteType::Memory] {
            let storage = self.get_storage(&storage_type);
            if let Some(route) = storage.load_route(id).await? {
                return Ok(Some(route));
            }
        }
        
        Ok(None)
    }
    
    pub async fn migrate_route(&self, id: i64, from_type: RouteType, to_type: RouteType) -> Result<(), StorageError> {
        // 从源存储加载
        let from_storage = self.get_storage(&from_type);
        let route = from_storage.load_route(id).await?
            .ok_or_else(|| StorageError::NotFound(format!("Route {} not found in {} storage", id, from_type)))?;
        
        // 保存到目标存储
        let to_storage = self.get_storage(&to_type);
        to_storage.save_route(&route).await?;
        
        // 从源存储删除
        from_storage.delete_route(id).await?;
        
        Ok(())
    }
}
```
