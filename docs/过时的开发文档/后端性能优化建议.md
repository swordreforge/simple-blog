### 1. 同步I/O阻塞异步运行时✅

**位置**: `src/handlers/api_handlers/filemanager.rs:336-366`

**问题描述**:
```rust
// 问题代码：在异步函数中使用阻塞式文件 I/O
if let Some(mut file) = file_option {
    use std::io::Read;
    let mut buffer = Vec::new();
    file.read_to_end(&mut buffer)?;
    
    if let Err(e) = std::fs::write(&file_path, &buffer) {
        eprintln!("保存文件失败: {}", e);
    }

```

**性能影响**: 阻塞整个tokio运行时，影响所有并发请求，在文件上传/下载场景下特别明显

**优化方案**:
```rust
// 使用异步文件操作
use tokio::fs;
use tokio::io::AsyncReadExt;

// 替换 std::fs::write
fs::write(&file_path, &buffer).await?;

// 替换 std::io::Read
let mut buffer = Vec::new();
let mut reader = file_option.unwrap();
reader.read_to_end(&mut buffer).await?;

// 对于大文件，使用流式处理
use tokio::io::{AsyncBufReadExt, BufReader};

let file = fs::File::open(&file_path).await?;
let reader = BufReader::new(file);
let mut lines = reader.lines();
while let Some(line) = lines.next_line().await? {
    // 处理每一行
}
```

**预期提升**: 50-80%（I/O密集型场景）

```rust

let attachments = match attachment_repo.get_by_passage_uuids(vec![uuid.clone()]).await {
```

**性能影响**: 每个附件都需要单独查询和删除，在多附件场景下性能急剧下降

**优化方案**:
```rust
// 批量查询所有附件
let passage_uuids: Vec<String> = passage_list.iter()
    .map(|p| p.uuid.clone())
    .collect();
let attachments = attachment_repo.get_by_passage_uuids(passage_uuids).await?;

// 批量删除文件（并行）
use futures::future;

let delete_tasks: Vec<_> = attachments.iter()
    .map(|attachment| {
        let path = attachment.file_path.clone();
        tokio::spawn(async move {
            fs::remove_file(&path).await.ok();
        })
    })
    .collect();

// 等待所有删除任务完成
let results = future::join_all(delete_tasks).await;
let deleted_files = results.len();
```

**预期提升**: 70-95%（关联数据场景）

**实施难度**: 低

**性能影响**: COUNT查询在全表扫描时很慢，尤其在有大量数据时

**优化方案**:

**方案1: 实现总数缓存**

```rust
use std::time::Duration;

// 缓存键生成
let cache_key = format!("passage:count:{}", date_filter);

// 尝试从缓存获取
if let Some(cached) = state.cache.manager().get(&cache_key).await {
    if let Ok(total) = cached.parse::<i64>() {
        return total;
    }
}

// 缓存未命中，执行查询
let total = if year.is_some() || month.is_some() || day.is_some() {
    passage_repo.count_published_by_date(year, month, day).await?
} else {
    passage_repo.count_published().await?
};

// 写入缓存（TTL: 10分钟）
state.cache.manager().set(
    &cache_key, 
    &total.to_string(), 
    Some(Duration::from_secs(600))
).await;

Ok(total)
```

**方案2: 优先使用游标分页**
```rust
// 避免COUNT查询，使用游标分页
pub async fn list_with_cursor(
    &self,
    cursor: Option<String>,
    limit: i64,
) -> Result<(Vec<Passage>, Option<String>, bool), Box<dyn Error>> {
    let limit = limit.min(100); // 限制最大每页数量
    let cursor_id = cursor.and_then(|c| c.parse::<i64>().ok());
    
    let mut query = "SELECT * FROM passages WHERE status = 'published'".to_string();
    
    if let Some(id) = cursor_id {
        query.push_str(&format!(" AND id > {}", id));
    }
    
    query.push_str(&format!(" ORDER BY id ASC LIMIT {}", limit + 1));
    
    let passages = self.query(&query, vec![]).await?;
    
    let has_more = passages.len() > limit as usize;
    let data = if has_more {
        passages.into_iter().take(limit as usize).collect()
    } else {
        passages
    };
    
    let next_cursor = data.last().map(|p| p.id.to_string());
    
    Ok((data, next_cursor, has_more))
}
```

```rust
let cache_manager = Arc::new(CacheManager::new(config));

// 2. 使用Cow避免不必要的字符串复制
use std::borrow::Cow;

fn format_name(name: &str) -> Cow<'_, str> {
    if name.contains(' ') {
        Cow::Owned(name.replace(' ', "_"))
    } else {
        Cow::Borrowed(name)
    }

// 3. 使用 format! 宏批量格式化
fn build_cache_key(prefix: &str, params: &[(&str, i32)]) -> String {
    let mut key = String::with_capacity(prefix.len() + params.len() * 20);
    key.push_str(prefix);
    for (k, v) in params {
        key.push_str(&format!(":{}:{}", k, v));
    }
    key
}

// 4. 对于日期格式化，预分配容量
fn format_datetime(dt: &chrono::DateTime<chrono::Local>) -> String {
    let mut result = String::with_capacity(19);
    result.push_str(&dt.format("%Y-%m-%d %H:%M:%S").to_string());
    result
```

### 5. 缓存失效过于激进

**位置**: `src/handlers/api_handlers/passage/query_handlers.rs:150-180`

**问题描述**:
```rust

    crate::cache::invalidate_all_passage_cache(state.cache.manager()).await;
}
```

**性能影响**: 一篇文章更新会清除所有列表缓存，导致缓存命中率降低

**优化方案**:
```rust
// 细粒度缓存失效
async fn invalidate_passage_cache(
    cache: &CacheManager,
    passage_uuid: &str,
    category: Option<&str>,
    date: Option<(&i32, Option<&i32>, Option<&i32>)>,
) {
    let mut keys_to_invalidate = vec![
        // 清除文章详情缓存
        format!("passage:detail:{}", passage_uuid),
        
        // 清除列表缓存
        "passage:list:all".to_string(),
        "passage:list:count:all".to_string(),
    ];
    
    // 清除分类列表缓存
    if let Some(cat) = category {
        keys_to_invalidate.push(format!("passage:list:category:{}", cat));
        keys_to_invalidate.push(format!("passage:list:count:category:{}", cat));
    }
    
    // 清除日期列表缓存
    if let Some((year, month, day)) = date {
        keys_to_invalidate.push(format!("passage:list:date:{}:count", year));
        if let Some(m) = month {
            keys_to_invalidate.push(format!("passage:list:date:{}-{}:count", year, m));
        }
    }
    
    // 批量删除缓存
    for key in keys_to_invalidate {
        cache.delete(&key).await.ok();
    }
}

// 更新时调用
async fn update_passage(
    state: &AppState,
    uuid: &str,
    title: String,
    content: String,
    category: String,
) -> Result<Passage, AppError> {
    // ... 更新逻辑 ...
    
    // 细粒度失效缓存
    let date = passage.published_at.map(|dt| {
        let y = dt.year();
        let m = Some(dt.month());
        let d = Some(dt.day());
        (&y, m.as_ref(), d.as_ref())
    });
    
    invalidate_passage_cache(
        state.cache.manager(),
        uuid,
        Some(&category),
        date
    ).await;
    
    Ok(passage)
}
```

```rust
// 记录慢查询
pub async fn query_with_monitor(
    &self,
    sql: &str,
    params: &[&dyn rusqlite::ToSql],
) -> Result<Vec<Row>, rusqlite::Error> {
    let start = std::time::Instant::now();
    
    let result = self.query(sql, params).await;
    
    let duration = start.elapsed();
    if duration.as_millis() > 100 {
        log::warn!("Slow query ({}ms): {}", duration.as_millis(), sql);
    }
    
    result
}
```

### 2. 缓存命中率监控
```rust
pub struct CacheMetrics {
    pub hits: AtomicU64,
    pub misses: AtomicU64,
}

impl CacheMetrics {
    pub fn hit_rate(&self) -> f64 {
        let hits = self.hits.load(Ordering::Relaxed);
        let misses = self.misses.load(Ordering::Relaxed);
        if hits + misses == 0 {
            0.0
        } else {
            hits as f64 / (hits + misses) as f64
        }
    }
}
```

### 4. 请求监控
```rust
pub struct RequestMetrics {
    pub total_requests: AtomicU64,
    pub active_requests: AtomicU64,
    pub avg_response_time: AtomicU64,
}

impl RequestMetrics {
    pub fn record_request(&self, duration: Duration) {
        self.total_requests.fetch_add(1, Ordering::Relaxed);
        
        let current_avg = self.avg_response_time.load(Ordering::Relaxed);
        let new_avg = (current_avg * 9 + duration.as_millis() as u64) / 10;
        self.avg_response_time.store(new_avg, Ordering::Relaxed);
    }
}
```
