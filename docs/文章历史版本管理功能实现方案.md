## 1. 概述

实现类似 Git 的文章版本历史管理功能，支持版本快照、差异对比、版本恢复、撤销/重做等核心功能。

### 1.1 设计目标

- **Git 风格的版本管理**：提供类似 Git 的版本控制体验
- **文件系统存储**：使用文件系统树形结构存储历史版本
- **灵活的配置**：通过 settings 表管理功能开关和参数
- **高性能**：支持内容去重、增量存储
- **完整的功能**：支持 diff、restore、undo/redo 等操作

### 1.2 核心特性

- ✅ 版本快照自动保存
- ✅ 版本历史查看
- ✅ 版本差异对比（diff）
- ✅ 版本恢复（多种模式）
- ✅ 撤销/重做功能
- ✅ 内容去重
- ✅ 树形结构存储
- ✅ 元数据管理

### 2.1 目录结构

```
markdown/
├── 2026/
│   ├── 02/
│   │   └── 02/
│   │       └── test-article-01.md          # 当前版本
│   └── 03/
│       └── 12/
│           └── ddd.md                      # 当前版本
│
└── .history/                              # 历史版本根目录
    ├── passages/                          # 文章版本历史
    │   └── {passage_uuid}/                 # 按文章 UUID 组织
    │       ├── v1_20260325-100000.md       # 版本 1
    │       ├── v2_20260325-110000.md       # 版本 2
    │       ├── v3_20260325-120000.md       # 版本 3
    │       └── .metadata.json              # 版本元数据
    │
    └── .index/                             # 全局索引（可选）
        └── {passage_uuid}.json             # 版本索引文件
```

### 2.2 文件命名规则

- **版本文件**：`v{version_number}_{timestamp}.md`
  - 示例：`v1_20260325-100000.md`
- **元数据文件**：`.metadata.json`
  - 存储在文章版本目录下
- **索引文件**：`{passage_uuid}.json`
  - 存储在 `.index` 目录下，用于快速查询

## 3. 数据库设计

### 3.1 版本历史表

```sql
CREATE TABLE IF NOT EXISTS passage_versions (
    id INTEGER PRIMARY KEY,
    passage_id INTEGER NOT NULL,
    passage_uuid TEXT NOT NULL,
    version_number INTEGER NOT NULL,
    
    -- 文件信息
    file_path TEXT NOT NULL,              -- 历史文件的完整路径
    file_size INTEGER,                    -- 文件大小（字节）
    file_hash TEXT,                       -- 文件内容哈希（用于去重）
    
    -- 只存储原始内容和元数据，不存储派生数据
    title TEXT NOT NULL,
    content TEXT NOT NULL,                -- 原始 Markdown 内容
    tags TEXT,                            -- JSON 数组字符串
    category TEXT,
    cover_image TEXT,
    
    -- 变更信息
    change_type TEXT NOT NULL,            -- auto, manual, restore, pre_restore
    change_reason TEXT,
    
    -- 操作信息
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    created_by TEXT DEFAULT 'system',
    
    -- Git 风格的树形结构
    parent_version_id INTEGER,            -- 父版本 ID
    branch_name TEXT,                     -- 分支名称（预留）
    
    FOREIGN KEY (passage_id) REFERENCES passages(id) ON DELETE CASCADE,
    FOREIGN KEY (parent_version_id) REFERENCES passage_versions(id) ON DELETE SET NULL,
    UNIQUE(passage_id, version_number)
);

-- 索引
CREATE INDEX IF NOT EXISTS idx_passage_versions_passage_id ON passage_versions(passage_id);
CREATE INDEX IF NOT EXISTS idx_passage_versions_uuid ON passage_versions(passage_uuid);
CREATE INDEX IF NOT EXISTS idx_passage_versions_file_hash ON passage_versions(file_hash);
CREATE INDEX IF NOT EXISTS idx_passage_versions_created ON passage_versions(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_passage_versions_parent ON passage_versions(parent_version_id);
CREATE INDEX IF NOT EXISTS idx_passage_versions_type ON passage_versions(change_type);
```

**重要说明**：

- **不存储派生数据**：`summary`、`html_content` 等派生数据不存储在版本历史表中
- **恢复时重新生成**：恢复版本时，使用原始 Markdown 内容重新生成 HTML 和摘要
- **数据一致性**：确保所有版本使用最新的转换逻辑，避免历史数据过时

    /// 获取最新版本号
    pub async fn get_latest_version_number(&self, passage_id: i64) -> Result<i32> {
        let pool = self.pool.clone();
        
        tokio::task::spawn_blocking(move || {
            let conn = pool.get()?;
            let version_number: i32 = conn.query_row(
                "SELECT COALESCE(MAX(version_number), 0) FROM passage_versions WHERE passage_id = ?1",
                params![passage_id],
                |row| row.get(0),
            )?;
            Ok(version_number)
        })
        .await?
    }

    /// 获取最新版本
    pub async fn get_latest_version(&self, passage_id: i64) -> Result<Option<PassageVersion>> {
        let latest_version_number = self.get_latest_version_number(passage_id).await?;
        if latest_version_number == 0 {
            return Ok(None);
        }
        self.get_by_version_number(passage_id, latest_version_number).await
    }

    /// 限制版本数量（删除旧版本）
    pub async fn trim_old_versions(&self, passage_id: i64, max_versions: usize) -> Result<()> {
        if max_versions == 0 {
            return Ok(()); // 0 表示不限制
        }
        
        let pool = self.pool.clone();
        
        tokio::task::spawn_blocking(move || {
            let conn = pool.get()?;
            
            // 获取当前版本总数
            let count: i64 = conn.query_row(
                "SELECT COUNT(*) FROM passage_versions WHERE passage_id = ?1",
                params![passage_id],
                |row| row.get(0),
            )?;
            
            if count as usize > max_versions {
                let to_delete = count as usize - max_versions;
                
                // 获取要删除的版本 ID 和文件路径
                let mut stmt = conn.prepare(
                    "SELECT id, file_path FROM passage_versions 
                     WHERE passage_id = ?1 
                     ORDER BY created_at ASC 
                     LIMIT ?2"
                )?;
                
                let versions_to_delete = stmt
                    .query_map(params![passage_id, to_delete as i64], |row| {
                        Ok((row.get::<_, i64>(0)?, row.get::<_, String>(1)?))
                    })?
                    .collect::<Result<Vec<_>, _>>()?;
                
                // 删除数据库记录
                conn.execute(
                    "DELETE FROM passage_versions 
                     WHERE passage_id = ?1 
                     AND id IN (
                         SELECT id FROM passage_versions 
                         WHERE passage_id = ?1 
                         ORDER BY created_at ASC 
                         LIMIT ?2
                     )",
                    params![passage_id, to_delete as i64],
                )?;
                
                // 删除历史文件
                for (_, file_path) in versions_to_delete {
                    let path = PathBuf::from(&file_path);
                    if path.exists() {
                        fs::remove_file(&path)?;
                    }
                }
            }
            
            Ok(())
        })
        .await?
    }

    /// 删除文章的所有版本
    pub async fn delete_by_passage_id(&self, passage_id: i64) -> Result<()> {
        let pool = self.pool.clone();
        
        tokio::task::spawn_blocking(move || {
            let conn = pool.get()?;
            
            // 获取所有版本文件路径
            let mut stmt = conn.prepare(
                "SELECT file_path FROM passage_versions WHERE passage_id = ?1"
            )?;
            
            let file_paths = stmt
                .query_map(params![passage_id], |row| {
                    Ok(row.get::<_, String>(0)?)
                })?
                .collect::<Result<Vec<_>, _>>()?;
            
            // 删除数据库记录
            conn.execute(
                "DELETE FROM passage_versions WHERE passage_id = ?1",
                params![passage_id],
            )?;
            
            // 删除历史文件
            for file_path in file_paths {
                let path = PathBuf::from(&file_path);
                if path.exists() {
                    fs::remove_file(&path)?;
                }
            }
            
            // 删除文章的版本目录
            // 需要获取 passage_uuid，这里简化处理
            
            Ok(())
        })

    /// 检查重复内容
    pub async fn check_duplicate_content(
        &self,
        passage_id: i64,

        let pool = self.pool.clone();
        let file_hash = file_hash.to_string();
        
        tokio::task::spawn_blocking(move || {

            let mut stmt = conn.prepare(

                 FROM passage_versions
                 WHERE passage_id = ?1 AND file_hash = ?2
                 ORDER BY created_at DESC
                 LIMIT 1"
            )?;
            
            let version = stmt
                .query_row(params![passage_id, file_hash], |row| {
                    Ok(PassageVersion {

    /// 保存版本（核心功能）
    pub async fn save_version(
        &self,
        passage_id: i64,
        passage_uuid: &str,
        passage: &Passage,
        change_type: &str,
        change_reason: Option<String>,
    ) -> Result<i64> {
        // 1. 获取配置
        let config = self.load_history_config().await?;
        
        if !config.enabled {
            return Ok(0); // 未启用历史功能
        }
        
        // 2. 计算文件哈希（用于去重）
        let file_hash = self.compute_hash(&passage.content);
        
        // 3. 检查是否已存在相同内容的版本
        if config.enable_deduplication {
            if let Some(existing) = self.version_repo
                .check_duplicate_content(passage_id, &file_hash)
                .await?
            {
                return Ok(existing.id.unwrap()); // 返回已存在的版本 ID
            }
        }
        
        // 4. 生成版本号
        let version_number = self.get_next_version_number(passage_id).await?;
        
        // 5. 根据存储模式处理
        let (file_path, file_size) = if config.storage_mode == "filesystem" {
            // 文件系统模式
            let history_file_path = self.generate_history_file_path(
                passage_uuid, 
                version_number
            ).await?;
            
            // 写入历史文件
            self.write_history_file(&history_file_path, &passage.content).await?;
            
            // 写入元数据文件
            self.write_metadata_file(
                passage_uuid, 
                version_number,
                passage,
                change_type,
                &change_reason
            ).await?;
            
            (history_file_path, passage.content.len() as i64)
        } else {
            // 数据库模式：文件路径为空
            (String::new(), passage.content.len() as i64)
        };
        
        // 6. 获取父版本 ID
        let parent_version_id = self.version_repo
            .get_latest_version(passage_id)
            .await?
            .and_then(|v| v.id);
        
        // 7. 保存到数据库
        let version = PassageVersion {
            id: None,
            passage_id,
            passage_uuid: passage_uuid.to_string(),
            version_number,
            file_path,
            file_size,
            file_hash: Some(file_hash),
            title: passage.title.clone(),
            content: passage.content.clone(),
            tags: passage.tags.clone(),
            category: passage.category.clone(),
            cover_image: passage.cover_image.clone(),
            change_type: change_type.to_string(),
            change_reason,
            created_at: Utc::now(),
            created_by: "system".to_string(),
            parent_version_id,
            branch_name: None,
        };
        
        let version_id = self.version_repo.create(&version).await?;
        
        // 8. 限制版本数量
        if config.max_versions > 0 {
            self.version_repo
                .trim_old_versions(passage_id, config.max_versions as usize)
                .await?;
        }
        
        // 9. 清除缓存
        self.clear_version_cache(passage_id).await;
        
        Ok(version_id)
    }

    /// 自动保存版本（在更新文章时调用）
    pub async fn auto_save_version(
        &self,
        passage_id: i64,
        passage_uuid: &str,
        old_passage: &Passage,
        new_passage: &Passage,
    ) -> Result<Option<i64>> {
        let config = self.load_history_config().await?;
        
        if !config.enabled {
            return Ok(None);
        }
        
        // 检查哪些字段发生了变化
        let mut changes = Vec::new();
        
        if config.save_on_title_change && old_passage.title != new_passage.title {
            changes.push("标题");
        }
        if config.save_on_content_change && old_passage.content != new_passage.content {
            changes.push("内容");
        }
        if config.save_on_tags_change && old_passage.tags != new_passage.tags {
            changes.push("标签");
        }
        if config.save_on_category_change && old_passage.category != new_passage.category {
            changes.push("分类");
        }
        if config.save_on_cover_change && old_passage.cover_image != new_passage.cover_image {
            changes.push("封面图片");
        }
        
        if changes.is_empty() {
            return Ok(None); // 没有需要保存的变更
        }
        
        let change_reason = Some(format!("自动保存：{}", changes.join("、")));
        
        let version_id = self.save_version(
            passage_id,
            passage_uuid,
            new_passage,
            "auto",
            change_reason,
        ).await?;
        
        Ok(Some(version_id))
    }

    /// 比较版本差异
    pub async fn diff_versions(
        &self,
        passage_id: i64,
        from_version: i32,
        to_version: i32,
    ) -> Result<VersionDiff> {
        let from = self.version_repo
            .get_by_version_number(passage_id, from_version)
            .await?
            .ok_or_else(|| anyhow!("版本 {} 不存在", from_version))?;
        
        let to = self.version_repo
            .get_by_version_number(passage_id, to_version)
            .await?
            .ok_or_else(|| anyhow!("版本 {} 不存在", to_version))?;
        
        let title_diff = DiffField {
            old: from.title.clone(),
            new: to.title.clone(),
            changed: from.title != to.title,
            diff_text: if from.title != to.title {
                Some(compute_line_diff(&from.title, &to.title))
            } else {
                None
            },
        };
        
        let content_diff = DiffField {
            old: from.content.clone(),
            new: to.content.clone(),
            changed: from.content != to.content,
            diff_text: if from.content != to.content {
                Some(compute_line_diff(&from.content, &to.content))
            } else {
                None
            },
        };
        
        let tags_diff = DiffField {
            old: from.tags.clone(),
            new: to.tags.clone(),
            changed: from.tags != to.tags,
            diff_text: if from.tags != to.tags {
                Some(compute_line_diff(&from.tags, &to.tags))
            } else {
                None
            },
        };
        
        let category_diff = DiffField {
            old: from.category.clone(),
            new: to.category.clone(),
            changed: from.category != to.category,
            diff_text: if from.category != to.category {
                Some(compute_line_diff(&from.category, &to.category))
            } else {
                None
            },
        };
        
        let cover_diff = DiffField {
            old: from.cover_image.clone(),
            new: to.cover_image.clone(),
            changed: from.cover_image != to.cover_image,
            diff_text: None, // 图片路径不需要行级 diff
        };
        
        Ok(VersionDiff {
            from_version,
            to_version,
            diff: FieldDiff {
                title: title_diff,
                content: content_diff,
                tags: tags_diff,
                category: category_diff,
                cover_image: cover_diff,
            },
        })
    }

    /// 恢复到某个版本
    pub async fn restore_version(
        &self,
        passage_id: i64,
        version_number: i32,
        change_reason: &str,
        mode: RestoreMode,
    ) -> Result<Passage> {
        // 1. 获取目标版本
        let version = self.version_repo
            .get_by_version_number(passage_id, version_number)
            .await?
            .ok_or_else(|| anyhow!("版本 {} 不存在", version_number))?;
        
        // 2. 获取当前文章
        let current_passage = self.passage_repo
            .get_by_id(passage_id)
            .await?
            .ok_or_else(|| anyhow!("文章不存在"))?;
        
        match mode {
            RestoreMode::Soft => {
                // 软恢复：只覆盖当前文件，不创建新版本
                let config = self.load_history_config().await?;
                
                if config.storage_mode == "filesystem" {
                    // 读取历史文件（原始 Markdown）
                    let markdown_content = self.read_history_file(&version.file_path).await?;
                    
                    // 覆盖当前文件
                    if let Some(ref current_file_path) = current_passage.file_path {
                        fs::write(current_file_path, &markdown_content).await?;
                    }
                }
                
                Ok(current_passage)
            }
            
            RestoreMode::Hard => {
                // 硬恢复：创建新版本记录，更新数据库
                
                // 3. 保存当前版本到历史（在恢复前）
                self.save_version(
                    passage_id,
                    &version.passage_uuid,
                    &current_passage,
                    "pre_restore",
                    Some(format!("恢复前的快照（恢复到版本 {}）", version_number))
                ).await?;
                
                // 4. 读取历史文件（原始 Markdown）
                let markdown_content = if self.load_history_config().await?.storage_mode == "filesystem" {
                    self.read_history_file(&version.file_path).await?
                } else {
                    version.content.clone()
                };
                
                // 5. 更新当前文件
                if let Some(ref current_file_path) = current_passage.file_path {
                    fs::write(current_file_path, &markdown_content).await?;
                }
                
                // 6. 重新生成派生数据（关键步骤！）
                let html_content = self.passage_service.convert_markdown_to_html(&markdown_content)?;
                let summary = self.passage_service.generate_summary(&markdown_content)?;
                
                // 7. 更新数据库记录
                let mut updated_passage = current_passage.clone();
                updated_passage.title = version.title.clone();
                updated_passage.content = markdown_content.clone();
                updated_passage.original_content = Some(markdown_content);
                updated_passage.html_content = Some(html_content);  // 重新生成
                updated_passage.summary = Some(summary);              // 重新生成
                updated_passage.tags = version.tags.clone();
                updated_passage.category = version.category.clone();
                updated_passage.cover_image = version.cover_image.clone();
                updated_passage.updated_at = Utc::now();
                
                // 8. 保存到数据库
                self.passage_repo.update(&updated_passage).await?;
                
                // 9. 保存恢复操作为新版本
                self.save_version(
                    passage_id,
                    &version.passage_uuid,
                    &updated_passage,
                    "restore",
                    Some(change_reason.to_string())
                ).await?;
                
                // 10. 清除缓存
                self.clear_passage_cache(passage_id).await;
                
                Ok(updated_passage)
            }
            
            RestoreMode::HardWithBackup => {
                // 硬恢复 + 备份：保留更多元数据
                self.restore_version(passage_id, version_number, change_reason, RestoreMode::Hard).await
            }
        }
    }

    /// 撤销最后一次更改
    pub async fn undo_last_change(&self, passage_id: i64) -> Result<Passage> {
        let config = self.load_history_config().await?;
        
        if !config.enable_undo_redo {
            return Err(anyhow!("撤销/重做功能未启用"));
        }
        
        // 1. 获取最新版本
        let latest_version = self.version_repo
            .get_latest_version(passage_id)
            .await?
            .ok_or_else(|| anyhow!("没有可用的版本"))?;
        
        // 2. 检查是否可以撤销
        if latest_version.parent_version_id.is_none() {
            return Err(anyhow!("没有可以撤销的版本"));
        }
        
        // 3. 获取父版本
        let parent_version = self.version_repo
            .get_by_id(latest_version.parent_version_id.unwrap())
            .await?
            .ok_or_else(|| anyhow!("父版本不存在"))?;
        
        // 4. 恢复到父版本
        self.restore_version(
            passage_id,
            parent_version.version_number,
            &format!("撤销操作：{}", latest_version.change_reason.unwrap_or_default()),
            RestoreMode::Hard
        ).await
    }

    async fn write_history_file(&self, path: &str, content: &str) -> Result<()> {
        let path = PathBuf::from(path);
        
        // 确保目录存在
        if let Some(parent) = path.parent() {
            fs::create_dir_all(parent).await?;

        fs::write(&path, content).await?;

    async fn read_history_file(&self, path: &str) -> Result<String> {
        let content = fs::read_to_string(path).await
            .map_err(|e| anyhow!("读取历史文件失败: {}", e))?;

    }

    /// 工具方法：写入元数据文件
    async fn write_metadata_file(
        &self,
        passage_uuid: &str,
        version_number: i32,
        passage: &Passage,
        change_type: &str,
        change_reason: &Option<String>,
    ) -> Result<()> {
        let config = self.load_history_config().await?;
        
        let metadata_path = PathBuf::from(&config.history_dir)
            .join("passages")
            .join(passage_uuid)
            .join(".metadata.json");
        
        let metadata = serde_json::json!({
            "version_number": version_number,
            "title": passage.title,
            "tags": passage.tags,
            "category": passage.category,
            "change_type": change_type,
            "change_reason": change_reason,
            "created_at": Utc::now().to_rfc3339(),
        });
        
        // 读取现有元数据
        let mut all_metadata: serde_json::Value = if metadata_path.exists() {
            let content = fs::read_to_string(&metadata_path).await?;
            serde_json::from_str(&content)?
        } else {
            serde_json::json!({})
        };
        
        // 添加新版本元数据
        all_metadata[format!("v{}", version_number)] = metadata;
        
        // 写入文件
        fs::write(
            &metadata_path,
            serde_json::to_string_pretty(&all_metadata)?
        ).await?;
        
        Ok(())
    }

    /// 工具方法：计算文件哈希
    fn compute_hash(&self, content: &str) -> String {
        use sha2::{Sha256, Digest};
        let mut hasher = Sha256::new();
        hasher.update(content.as_bytes());
        format!("{:x}", hasher.finalize())
    }

    }
}

/// 使用 similar crate 计算行级差异
fn compute_line_diff(old: &str, new: &str) -> String {
    let diff = TextDiff::from_lines(old, new);
    
    let mut result = String::new();
    for op in diff.ops() {
        for change in diff.iter_changes(op) {
            let sign = match change.tag() {
                ChangeTag::Delete => "-",
                ChangeTag::Insert => "+",
                ChangeTag::Equal => " ",
            };
            result.push_str(&format!("{}{}", sign, change.value()));
        }
    }
    result
}

### 7.4 API 权限控制

#### 7.4.1 公开 API

- **无需认证**：所有公开 API 都可以匿名访问
- **只读操作**：仅支持查询操作，不支持修改
- **缓存策略**：使用缓存减轻服务器压力
- **限流保护**：建议添加限流机制防止滥用

#### 7.4.2 管理 API

- **需要认证**：所有管理 API 都需要用户认证
- **权限检查**：
  - 作者：可以管理自己创建的文章版本
  - 管理员：可以管理所有文章版本
- **操作日志**：记录所有管理操作，便于审计
- **安全验证**：使用 CSRF Token 防止跨站请求伪造

/// 构建版本树
fn build_version_tree(versions: &[PassageVersion]) -> Option<serde_json::Value> {
    if versions.is_empty() {
        return None;
    }
    
    // 创建版本映射
    let version_map: std::collections::HashMap<i64, &PassageVersion> = versions
        .iter()
        .filter_map(|v| v.id.map(|id| (id, v)))
        .collect();
    
    // 找到根版本（没有父版本的版本）
    let root = versions.iter().find(|v| v.parent_version_id.is_none())?;
    
    // 递归构建树
    Some(build_tree_node(root, &version_map))
}

fn build_tree_node(
    version: &PassageVersion,
    version_map: &std::collections::HashMap<i64, &PassageVersion>,
) -> serde_json::Value {
    let children: Vec<serde_json::Value> = version_map
        .values()
        .filter(|v| v.parent_version_id == version.id)
        .map(|v| build_tree_node(v, version_map))
        .collect();
    
    serde_json::json!({
        "id": version.id,
        "version_number": version.version_number,
        "title": version.title,
        "change_type": version.change_type,
        "created_at": version.created_at,
        "children": children
    })
}

    match tokio::task::spawn_blocking(move || {
        let conn = pool.get()?;
        
        // 更新每个提供的字段
        if let Some(enabled) = updates.get("enabled").and_then(|v| v.as_bool()) {
            SettingRepository::set(&conn, &Setting {
                key: "passage_history.enabled".to_string(),
                value: enabled.to_string(),
                r#type: "bool".to_string(),
                description: Some("是否启用文章历史版本管理".to_string()),
                category: "passage".to_string(),
                created_at: Utc::now(),
                updated_at: Utc::now(),
            })?;
        }
        
        if let Some(storage_mode) = updates.get("storage_mode").and_then(|v| v.as_str()) {
            SettingRepository::set(&conn, &Setting {
                key: "passage_history.storage_mode".to_string(),
                value: storage_mode.to_string(),
                r#type: "string".to_string(),
                description: Some("存储模式".to_string()),
                category: "passage".to_string(),
                created_at: Utc::now(),
                updated_at: Utc::now(),
            })?;
        }
        
        if let Some(history_dir) = updates.get("history_dir").and_then(|v| v.as_str()) {
            SettingRepository::set(&conn, &Setting {
                key: "passage_history.history_dir".to_string(),
                value: history_dir.to_string(),
                r#type: "string".to_string(),
                description: Some("历史版本存储目录".to_string()),
                category: "passage".to_string(),
                created_at: Utc::now(),
                updated_at: Utc::now(),
            })?;
        }
        
        if let Some(max_versions) = updates.get("max_versions").and_then(|v| v.as_i64()) {
            SettingRepository::set(&conn, &Setting {
                key: "passage_history.max_versions".to_string(),
                value: max_versions.to_string(),
                r#type: "number".to_string(),
                description: Some("保留历史版本的最大数量".to_string()),
                category: "passage".to_string(),
                created_at: Utc::now(),
                updated_at: Utc::now(),
            })?;
        }
        
        // 其他字段类似...
        
        let settings = crate::templates::load_passage_history_settings(&conn)?;
        Ok::<_, Box<dyn std::error::Error>>(settings)
    })
    .await
    {
        Ok(Ok(settings)) => HttpResponse::Ok().json(serde_json::json!({
            "success": true,
            "settings": settings
        })),
        Ok(Err(e)) => HttpResponse::InternalServerError().json(serde_json::json!({
            "error": format!("更新配置失败: {}", e)
        })),
        Err(e) => HttpResponse::InternalServerError().json(serde_json::json!({
            "error": format!("任务执行失败: {}", e)
        })),
    }
}

/// 加载文章历史版本配置
pub fn load_passage_history_settings(
    conn: &rusqlite::Connection,
) -> Result<PassageHistorySettings, Box<dyn std::error::Error>> {
    let keys = vec![
        "passage_history.enabled",
        "passage_history.storage_mode",
        "passage_history.history_dir",
        "passage_history.max_versions",
        "passage_history.enable_deduplication",
        "passage_history.save_on_title_change",
        "passage_history.save_on_content_change",
        "passage_history.save_on_tags_change",
        "passage_history.save_on_summary_change",
        "passage_history.save_on_category_change",
        "passage_history.save_on_cover_change",
        "passage_history.enable_undo_redo",
    ];

    let mut settings: std::collections::HashMap<String, String> = std::collections::HashMap::new();

    for key in keys {
        if let Some(setting) = SettingRepository::get(conn, key)? {
            settings.insert(key.to_string(), setting.value);
        }
    }

    let mut config = PassageHistorySettings::default();

    for (key, value) in settings {
        match key.as_str() {
            "passage_history.enabled" => {
                config.enabled = value.parse().unwrap_or(true);
            }
            "passage_history.storage_mode" => {
                config.storage_mode = value;
            }
            "passage_history.history_dir" => {
                config.history_dir = value;
            }
            "passage_history.max_versions" => {
                config.max_versions = value.parse().unwrap_or(50);
            }
            "passage_history.enable_deduplication" => {
                config.enable_deduplication = value.parse().unwrap_or(true);
            }
            "passage_history.save_on_title_change" => {
                config.save_on_title_change = value.parse().unwrap_or(true);
            }
            "passage_history.save_on_content_change" => {
                config.save_on_content_change = value.parse().unwrap_or(true);
            }
            "passage_history.save_on_tags_change" => {
                config.save_on_tags_change = value.parse().unwrap_or(true);
            }
            "passage_history.save_on_summary_change" => {
                config.save_on_summary_change = value.parse().unwrap_or(true);
            }
            "passage_history.save_on_category_change" => {
                config.save_on_category_change = value.parse().unwrap_or(false);

            "passage_history.save_on_cover_change" => {
                config.save_on_cover_change = value.parse().unwrap_or(false);

                config.enable_undo_redo = value.parse().unwrap_or(true);

            _ => {}
        }
    }

    Ok(config)

// 渲染差异视图
function renderDiffView(diff) {
  const container = document.getElementById('diff-view');
  
  Object.entries(diff).forEach(([field, data]) => {
    if (data.changed) {
      const fieldContainer = document.createElement('div');
      fieldContainer.className = 'diff-field';
      
      fieldContainer.innerHTML = `
        <h3>${field}</h3>
        ${data.diff_text ? `
          <pre class="diff-content">${highlightDiff(data.diff_text)}</pre>
        ` : `
          <div class="diff-simple">
            <div class="diff-old">旧值: ${data.old}</div>
            <div class="diff-new">新值: ${data.new}</div>
          </div>
        `}
      `;
      
      container.appendChild(fieldContainer);
    }
  });
}

// 高亮差异
function highlightDiff(diffText) {
  return diffText
    .split('\n')
    .map(line => {
      if (line.startsWith('-')) {
        return `<span class="diff-remove">${line}</span>`;
      } else if (line.startsWith('+')) {
        return `<span class="diff-add">${line}</span>`;
      } else {
        return `<span class="diff-context">${line}</span>`;
      }
    })
    .join('\n');
}

// 恢复版本
function restoreVersion(versionNumber) {
  const reason = prompt('请输入恢复原因:', `恢复到版本 ${versionNumber}`);
  if (!reason) return;
  
  const mode = confirm('是否创建新版本记录？\n确定 = 硬恢复（创建记录）\n取消 = 软恢复（仅覆盖文件）') 
    ? 'hard' 
    : 'soft';
  
  fetch(`/api/passage/${currentUuid}/versions/${versionNumber}/restore`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      change_reason: reason,
      mode: mode
    })
  })
    .then(res => res.json())
    .then(data => {
      if (data.success) {
        alert('恢复成功！');
        location.reload();
      } else {
        alert('恢复失败: ' + data.error);
      }
    });
}

## 16. 注意事项

### 16.1 性能考虑

- **版本数量限制**：默认保留 50 个版本，可根据需要调整
- **内容去重**：相同内容不重复存储，节省空间
- **缓存策略**：版本列表和详情都需要缓存
- **文件系统操作**：使用异步文件操作，避免阻塞

### 16.2 权限控制

- 只有作者和管理员可以查看版本历史
- 恢复操作需要特殊权限
- 建议添加操作日志

### 16.3 存储优化

- 考虑使用压缩存储大文本内容
- 可以为版本表添加分区
- 定期清理过期的版本

### 16.4 用户体验

- 提供清晰的版本对比视图
- 支持一键恢复
- 显示修改原因和时间
- 支持撤销/重做操作

### 16.5 数据安全

- 历史文件独立存储，即使数据库损坏也能恢复
- 恢复前自动备份当前版本
- 建议定期备份 `.history` 目录
